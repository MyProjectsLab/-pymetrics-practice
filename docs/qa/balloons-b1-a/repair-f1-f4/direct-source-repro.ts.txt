 installationId:'repro-installation',sessionId:'repro-session',chainId:'repro-chain',stage:'baseline',
 mapping:{'Type 1':'A','Type 2':'B','Type 3':'C'},exposureGeneration:0,ownershipGeneration:0,
},{wallTimeMs:0,monotonicMs:0,clockEpochId:'repro'})
function action(kind:CoreAction['kind']) {
 const t = transition(original,{kind,actionId:original.nextActionId,expectedRound:original.round,expectedRevision:original.revision,source:'keyboard',time:{wallTimeMs:original.revision+1,monotonicMs:original.revision+1,clockEpochId:'repro'}})
 if (!t.accepted) throw new Error('unexpected rejection')
 original=t.state
}
while(original.status!=='completed') {
 while(original.status==='active' && original.current.attemptedPumps<6) action('pump')
 if(original.status==='active')action('collect')
 if(original.status==='resolved')action('next')
}
function refused(s:CoreState) { try{return basicReport(s).status!=='headless_complete'}catch{return true} }
test('control: genuine complete BL1 at k=6 is 810 banked and 125 lost',()=>{
 expect(basicReport(original)).toMatchObject({status:'headless_complete',total:{bankedCents:810,lostCents:125}})
})
